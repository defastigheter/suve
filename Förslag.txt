/* ========== Hjälpfunktioner och antaganden ==========
   - Grundantagande: 60 W/m² (0.06 kW/m²) för mellanzone.
   - Zonmultiplikatorer: north 1.20, middle 1.00, south 0.90.
   - Isoleringsfaktorer: 30-40cm -> 0.85, 20-30 -> 0.95, 10-20 -> 1.05, 0-10 -> 1.20
   - Fönsterfaktor: 3-glas 0.9, 2-glas 1.0, 1-glas 1.15
   - Rundning: uppåt till närmaste 0.5 kW vid rekommendation
*/
function getZoneMultiplier(z) {
  if (z === 'north') return 1.20;
  if (z === 'south') return 0.90;
  return 1.00;
}
function getInsulationFactor(bucket) {
  switch(bucket) {
    case '30-40': return 0.85;
    case '20-30': return 0.95;
    case '10-20': return 1.05;
    case '0-10':  return 1.20;
    default: return 1.0;
  }
}
function getWindowFactor(val) {
  if (val === '3') return 0.9;
  if (val === '1') return 1.15;
  return 1.0;
}
function roundUpToHalf(x) {
  return Math.ceil(x * 2) / 2;
}

/* Kör beräkning */
function runCalculation() {
  // Läs in värden (vi antar att användaren fyllt rimliga siffror)
  const zone = document.getElementById('zone').value;
  const area = parseFloat(document.getElementById('area').value) || 0;
  const basement = parseFloat(document.getElementById('basement').value) || 0;
  const attic = parseFloat(document.getElementById('attic').value) || 0;
  const windows = document.getElementById('windows').value;
  const insulation = document.getElementById('insulation').value;
  const desiredType = document.getElementById('desiredType').value;

  const totalArea = area + basement + attic;

  // Parametrar
  const baseWperM2 = 0.06; // kW per m2 = 60 W/m2
  const zoneM = getZoneMultiplier(zone);
  const insF = getInsulationFactor(insulation);
  const winF = getWindowFactor(windows);

  // Beräkning: effekt (kW)
  // effekt = area * base * zone * isolering * fönster
  const rawKW = totalArea * baseWperM2 * zoneM * insF * winF;
  const neededKW = Math.max(0.5, roundUpToHalf(rawKW)); // minst 0.5 kW
  const neededKWRounded = neededKW.toFixed(2);

  // Visa sammanfattning
  const summaryEl = document.getElementById('calcSummary');
  summaryEl.innerHTML = `Total uppvärmd yta: <strong>${totalArea} m²</strong> (huvudbyggnad ${area}m², källare ${basement}m², vind ${attic}m²). Grundantagande: 60 W/m² justerat för zon, isolering och fönster.`;

  document.getElementById('neededKW').innerText = neededKWRounded;

  // Hämta rekommendationer
  const recs = getRecommendations(neededKW, desiredType);

  const recContainer = document.getElementById('recommendations');
  recContainer.innerHTML = ''; // rensa

  if (recs.length === 0) {
    recContainer.innerHTML = `<p class="text-gray-700">Inga exakta matchningar hittades för det här kravet och typen. Överväg en större enhet eller kontakta oss för en platsbesiktning.</p>`;
  } else {
    const list = document.createElement('div');
    list.className = 'grid grid-cols-1 md:grid-cols-3 gap-3';
    recs.forEach(r => {
      const card = document.createElement('div');
      card.className = 'p-3 border rounded-lg bg-white';
      card.innerHTML = `
        <img src="${r.image}" alt="${r.title}" class="w-full h-28 object-contain mb-2" />
        <div class="text-sm font-semibold text-gray-900">${r.title}</div>
        <div class="text-xs text-gray-600">Typ: ${r.type}</div>
        <div class="mt-2 text-sm">Nom. eff: <strong>${r.nominal} kW</strong></div>
        <a href="${r.url}" class="mt-2 inline-block text-AccentR-500 text-sm font-semibold hover:underline">Produktinfo</a>
      `;
      list.appendChild(card);
    });
    recContainer.appendChild(list);
  }

  // Visa resultatkort
  document.getElementById('resultCard').classList.remove('hidden');

  // Spara senaste beräkningen i variabel ( används när användaren klickar "Skicka till offert" )
  window.lastCalc = {
    timestamp: new Date().toISOString(),
    zone, area, basement, attic, totalArea, windows, insulation, desiredType,
    neededKW: Number(neededKWRounded),
    recs
  };

  // Förbered mailto-länk (fallback)
  prepareMailto(window.lastCalc);
}

/* Enkel matchningsalgoritm:
   - Filtrera produkter enligt desiredType (eller alla)
   - Ta de produkter vars nominal (eller max) >= neededKW, annars välj de med störst nominal
   - Sortera efter hur nära nominal är till neededKW (minimera överdimensionering)
*/
function getRecommendations(neededKW, desiredType) {
  let candidates = PRODUCTS.slice();

  if (desiredType !== 'any') {
    candidates = candidates.filter(p => p.type === desiredType);
  }

  // Produkter som tekniskt klarar behov (nominal >= neededKW OR max >= neededKW)
  const fits = candidates.filter(p => {
    if (p.nominal && p.nominal >= neededKW) return true;
    if (p.max && p.max >= neededKW) return true;
    return false;
  });

  // Om vi hittar passande - rankera efter närhet (minimera nominal - needed)
  if (fits.length > 0) {
    fits.sort((a,b) => (a.nominal || a.max || 999) - (b.nominal || b.max || 999));
    return fits.slice(0,3);
  }

  // Annars returnera de tre mest kraftfulla i kategorin (för att visa alternativ)
  candidates.sort((a,b) => (b.nominal || b.max || 0) - (a.nominal || a.max || 0));
  return candidates.slice(0,3);
}

  document.getElementById('area').value = 100;
  document.getElementById('basement').value = 0;
  document.getElementById('attic').value = 0;
  document.getElementById('resultCard').classList.add('hidden');
  window.lastCalc = null;
}

function prepareMailto(calc) {
  try {
    const body = [
      `Offertförfrågan - värmepumpsberäkning`,
      ``,
      `Datum: ${new Date().toLocaleString()}`,
      `Zon: ${calc.zone}`,
      `Total yta: ${calc.totalArea} m² (huvud: ${calc.area}, källare: ${calc.basement}, vind: ${calc.attic})`,
      `Isolering: ${calc.insulation}`,
      `Fönster: ${calc.windows}-glas`,
      `Önskad typ: ${calc.desiredType}`,
      `Beräknat effektbehov: ${calc.neededKW} kW`,
      ``,
      `Rekommendationer: ${calc.recs.map(r => r.title + ' (' + r.nominal + ' kW)').join('; ')}`,
      ``,
      `Vänligen återkom med offert och installationsförslag.`
    ].join('\n');

    const mailto = `mailto:info@suve.se?subject=${encodeURIComponent('Offertförfrågan värmepump')}&body=${encodeURIComponent(body)}`;
    const a = document.getElementById('mailtoFallback');
    a.href = mailto;
    a.classList.remove('hidden');
    a.innerText = 'Skicka via e-post';
    a.onclick = function(){ /* låt det öppnas normalt */ };
  } catch(e) {
    console.warn('Kunde inte förbereda mailto', e);
  }
}
